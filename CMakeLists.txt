cmake_minimum_required(VERSION 3.21)
project(Lite LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(CEF_ROOT "${CMAKE_SOURCE_DIR}/vendor/cef" CACHE PATH "CEF SDK")
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
set(PROJECT_ARCH "${CMAKE_SYSTEM_PROCESSOR}" CACHE STRING "arm64 or x86_64")
set(USE_SANDBOX ON CACHE BOOL "Chromium sandbox" FORCE)
find_package(CEF REQUIRED)
# The SDK header is the single source for the engine version Lite shows and sends.
file(READ "${CEF_ROOT}/include/cef_version.h" CEF_VERSION_HEADER)
if(NOT CEF_VERSION_HEADER MATCHES "#define CEF_VERSION \"[^\"]*chromium-(([0-9]+)\\.[0-9.]+)\"")
  message(FATAL_ERROR "Could not read the Chromium version from cef_version.h")
endif()
set(LITE_CHROMIUM_VERSION "${CMAKE_MATCH_1}")
set(LITE_CHROMIUM_MAJOR "${CMAKE_MATCH_2}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CEF_ROOT}/include/cef_version.h")
set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0")
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
add_library(LiteCore STATIC lite/model/LTWebsiteData.m lite/model/LTModel.m lite/model/LTStore.m lite/model/LTCommandIndex.m lite/model/LTSearch.m lite/model/LTLoginStore.m lite/model/LTLoginVault.m lite/model/LTGitHub.m lite/model/LTExtensions.m lite/model/LTCrashReports.m lite/model/LTFilterUpdates.m lite/migration/LTImporter.m lite/migration/LTExporter.m lite/browser/LTContentBlocker.mm)
target_compile_options(LiteCore PRIVATE -fobjc-arc -Wall)
target_compile_definitions(LiteCore PRIVATE LITE_CHROMIUM_VERSION="${LITE_CHROMIUM_VERSION}")
target_link_libraries(LiteCore PUBLIC "-framework Foundation" "-framework Security" sqlite3)
target_sources(LiteCore PRIVATE lite/browser/LTCustomCosmetics.m)
# LiteTestApp is the same application plus the Chromium and native-window suites.
# Release bundles use Lite, which must not contain any test code or test flags.
set(LITE_APP_SOURCES lite/macos/LTDownloadHistory.m lite/macos/main.mm lite/macos/LTUpdater.m lite/macos/LTUI.m lite/macos/LTTheme.m lite/macos/LTSidebar.m lite/macos/LTWindow.mm lite/macos/LTCommandPanel.m lite/macos/LTLibraryPanel.m lite/macos/LTFaviconCache.m lite/macos/LTPasswords.m lite/macos/LTShortcuts.m lite/macos/LTTaskManager.m lite/macos/LTExtensionsPanel.m lite/browser/LTEngine.mm lite/performance/LTPerformance.m)
add_executable(Lite ${LITE_APP_SOURCES})
add_executable(LiteTestApp ${LITE_APP_SOURCES} lite/tests/LTSmoke.mm lite/tests/LTNativeSmoke.mm)
target_compile_definitions(LiteTestApp PRIVATE LITE_TEST_BUILD)
foreach(app Lite LiteTestApp)
  target_include_directories(${app} PRIVATE "${CEF_ROOT}")
  target_compile_definitions(${app} PRIVATE CEF_USE_SANDBOX LITE_CHROMIUM_MAJOR="${LITE_CHROMIUM_MAJOR}")
  target_compile_options(${app} PRIVATE -fobjc-arc -Wall -Wno-unused-parameter)
  target_link_libraries(${app} PRIVATE LiteCore libcef_dll_wrapper "-framework Cocoa" "-framework UniformTypeIdentifiers" "-framework QuartzCore" "-framework ImageIO" "-framework AVFoundation")
endforeach()
add_executable(LiteHelper lite/browser/helper.cc)
target_include_directories(LiteHelper PRIVATE "${CEF_ROOT}")
target_compile_definitions(LiteHelper PRIVATE CEF_USE_SANDBOX)
target_link_libraries(LiteHelper PRIVATE libcef_dll_wrapper "-framework Cocoa")
enable_testing()
add_executable(LiteTests lite/tests/tests.m lite/performance/LTPerformance.m lite/macos/LTFaviconCache.m lite/macos/LTShortcuts.m lite/macos/LTUI.m lite/macos/LTTheme.m)

target_link_libraries(LiteTests PRIVATE LiteCore "-framework Cocoa" "-framework QuartzCore" "-framework ImageIO")
add_test(NAME LiteCoreTests COMMAND LiteTests "${CMAKE_SOURCE_DIR}/lite/tests/fixtures")

target_compile_options(LiteProfileTests PRIVATE -fobjc-arc -Wall)
target_link_libraries(LiteProfileTests PRIVATE LiteCore)

target_compile_options(LiteDownloadHistoryTests PRIVATE -fobjc-arc -Wall)
target_link_libraries(LiteDownloadHistoryTests PRIVATE "-framework Cocoa")
add_test(NAME LiteDownloadHistoryTests COMMAND LiteDownloadHistoryTests)

target_include_directories(LiteBlockingTests PRIVATE "${CEF_ROOT}")
target_compile_options(LiteBlockingTests PRIVATE -fobjc-arc -Wall)

add_test(NAME LiteBlockingTests COMMAND LiteBlockingTests "${CMAKE_SOURCE_DIR}/resources/ContentBlocking")

# Sparkle is vendored from its upstream checksum-pinned binary distribution.
set(SPARKLE_FRAMEWORK "${CMAKE_SOURCE_DIR}/vendor/sparkle/Sparkle.framework")
if(NOT EXISTS "${SPARKLE_FRAMEWORK}/Sparkle")
  message(FATAL_ERROR "Run python3 script/fetch_sparkle.py before configuring Lite")
endif()
foreach(app Lite LiteTestApp)
  target_link_libraries(${app} PRIVATE "${SPARKLE_FRAMEWORK}")
  set_target_properties(${app} PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks")
endforeach()
add_executable(LiteUpdateTests lite/tests/update_tests.m lite/macos/LTUpdater.m)
target_compile_options(LiteUpdateTests PRIVATE -fobjc-arc -Wall)
target_link_libraries(LiteUpdateTests PRIVATE "-framework Cocoa" "${SPARKLE_FRAMEWORK}")
add_test(NAME LiteUpdateTests COMMAND LiteUpdateTests)
