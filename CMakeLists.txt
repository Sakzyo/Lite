cmake_minimum_required(VERSION 3.21)
project(Lite LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
set(CEF_ROOT "${CMAKE_SOURCE_DIR}/vendor/cef" CACHE PATH "CEF SDK")
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
set(PROJECT_ARCH "${CMAKE_SYSTEM_PROCESSOR}" CACHE STRING "arm64 or x86_64")
set(USE_SANDBOX ON CACHE BOOL "Chromium sandbox" FORCE)
find_package(CEF REQUIRED)
set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0")
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
add_library(LiteCore STATIC lite/model/LTWebsiteData.m lite/model/LTModel.m lite/model/LTStore.m lite/model/LTCommandIndex.m lite/model/LTLoginStore.m lite/model/LTLoginVault.m lite/model/LTGitHub.m lite/migration/LTImporter.m lite/browser/LTContentBlocker.mm)
target_compile_options(LiteCore PRIVATE -fobjc-arc -Wall)
target_link_libraries(LiteCore PUBLIC "-framework Foundation" "-framework Security" sqlite3)
add_executable(Lite lite/macos/LTDownloadHistory.m lite/macos/main.mm lite/macos/LTUpdater.m lite/macos/LTUI.m lite/macos/LTSidebar.m lite/macos/LTWindow.mm lite/macos/LTCommandPanel.m lite/macos/LTLibraryPanel.m lite/macos/LTFaviconCache.m lite/macos/LTPasswords.m lite/macos/LTShortcuts.m lite/macos/LTTaskManager.m lite/browser/LTEngine.mm lite/performance/LTPerformance.m lite/tests/LTSmoke.mm lite/tests/LTNativeSmoke.mm)
target_include_directories(Lite PRIVATE "${CEF_ROOT}")
target_compile_definitions(Lite PRIVATE CEF_USE_SANDBOX)
target_compile_options(Lite PRIVATE -fobjc-arc -Wall -Wno-unused-parameter)
target_link_libraries(Lite PRIVATE LiteCore libcef_dll_wrapper "-framework Cocoa" "-framework UniformTypeIdentifiers" "-framework QuartzCore" "-framework ImageIO")
add_executable(LiteHelper lite/browser/helper.cc)
target_include_directories(LiteHelper PRIVATE "${CEF_ROOT}")
target_compile_definitions(LiteHelper PRIVATE CEF_USE_SANDBOX)
target_link_libraries(LiteHelper PRIVATE libcef_dll_wrapper "-framework Cocoa")
enable_testing()
add_executable(LiteTests lite/tests/tests.m lite/performance/LTPerformance.m lite/macos/LTFaviconCache.m lite/macos/LTShortcuts.m lite/macos/LTUI.m)
target_compile_options(LiteTests PRIVATE -fobjc-arc)
target_link_libraries(LiteTests PRIVATE LiteCore "-framework Cocoa" "-framework ImageIO")
add_test(NAME LiteCoreTests COMMAND LiteTests "${CMAKE_SOURCE_DIR}/lite/tests/fixtures")
add_executable(LiteVaultTests lite/tests/login_vault_tests.m)
target_compile_options(LiteVaultTests PRIVATE -fobjc-arc -Wall)
target_link_libraries(LiteVaultTests PRIVATE LiteCore)
add_test(NAME LiteVaultTests COMMAND LiteVaultTests)
add_executable(LiteProfileTests lite/tests/profile_recovery_tests.m)
target_compile_options(LiteProfileTests PRIVATE -fobjc-arc -Wall)
target_link_libraries(LiteProfileTests PRIVATE LiteCore)
add_test(NAME LiteProfileTests COMMAND LiteProfileTests)
add_executable(LiteDownloadHistoryTests lite/tests/download_history_tests.m lite/macos/LTDownloadHistory.m)
target_compile_options(LiteDownloadHistoryTests PRIVATE -fobjc-arc -Wall)
target_link_libraries(LiteDownloadHistoryTests PRIVATE "-framework Cocoa")
add_test(NAME LiteDownloadHistoryTests COMMAND LiteDownloadHistoryTests)
add_executable(LiteBlockingTests lite/tests/content_blocking_tests.mm)
target_include_directories(LiteBlockingTests PRIVATE "${CEF_ROOT}")
target_compile_options(LiteBlockingTests PRIVATE -fobjc-arc -Wall)
target_link_libraries(LiteBlockingTests PRIVATE LiteCore)
add_test(NAME LiteBlockingTests COMMAND LiteBlockingTests "${CMAKE_SOURCE_DIR}/resources/ContentBlocking")

# Sparkle is vendored from its upstream checksum-pinned binary distribution.
set(SPARKLE_FRAMEWORK "${CMAKE_SOURCE_DIR}/vendor/sparkle/Sparkle.framework")
if(NOT EXISTS "${SPARKLE_FRAMEWORK}/Sparkle")
  message(FATAL_ERROR "Run python3 script/fetch_sparkle.py before configuring Lite")
endif()
target_link_libraries(Lite PRIVATE "${SPARKLE_FRAMEWORK}")
set_target_properties(Lite PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks")
add_executable(LiteUpdateTests lite/tests/update_tests.m lite/macos/LTUpdater.m)
target_compile_options(LiteUpdateTests PRIVATE -fobjc-arc -Wall)
target_link_libraries(LiteUpdateTests PRIVATE "-framework Cocoa" "${SPARKLE_FRAMEWORK}")
add_test(NAME LiteUpdateTests COMMAND LiteUpdateTests)

add_executable(LiteWebsiteDataTests lite/tests/website_data_tests.m)
target_compile_options(LiteWebsiteDataTests PRIVATE -fobjc-arc -Wall)
target_link_libraries(LiteWebsiteDataTests PRIVATE LiteCore)
add_test(NAME LiteWebsiteDataTests COMMAND LiteWebsiteDataTests)
